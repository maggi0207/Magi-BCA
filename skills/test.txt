-- 1) Recent NPP-type CVI rows on Credentialing
SELECT TOP 30 *
FROM VisualCACTUS.Credentialing
WHERE TYPE_RTK = 'HCA0095563'
   OR CREDENTIALINGSTATUS_RTK = 'UDP0000001'
ORDER BY 1 DESC;

-- 2) Same on group table
SELECT TOP 30 *
FROM VisualCACTUS.CREDENTIALINGGROUP
WHERE TYPE_RTK = 'HCA0095563'
ORDER BY 1 DESC;
